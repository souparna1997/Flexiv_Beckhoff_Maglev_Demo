cmake_minimum_required(VERSION 3.16)
project(Flexiv_Beckhoff_Maglev_Demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
find_package(CHAI3D REQUIRED)
foreach(component COMMON URDF MODEL SIMULATION GRAPHICS PRIMITIVES INTERFACES)
    find_package(SAI-${component} REQUIRED)
endforeach()
set(DEMO_ROBOT_FOLDER "${CMAKE_CURRENT_SOURCE_DIR}/../sra_project/sai_analysis/urdf/rizon4s"
    CACHE PATH "Folder containing the SRA Rizon model and Rizon4_meshes")
if(NOT EXISTS "${DEMO_ROBOT_FOLDER}/flexiv_Rizon4s_kinematics_with_grav_static.urdf")
    message(FATAL_ERROR "Set DEMO_ROBOT_FOLDER to the SRA Rizon model directory")
endif()
add_executable(maglev_opensai src/main.cpp)
target_include_directories(maglev_opensai PRIVATE ${CHAI3D_INCLUDE_DIRS})
add_definitions(${CHAI3D_DEFINITIONS} ${SAI-SIMULATION_DEFINITIONS} ${SAI-GRAPHICS_DEFINITIONS})
target_link_directories(maglev_opensai PRIVATE ${CHAI3D_LIBRARY_DIRS})
foreach(component INTERFACES PRIMITIVES SIMULATION GRAPHICS MODEL URDF COMMON)
    target_include_directories(maglev_opensai PRIVATE ${SAI-${component}_INCLUDE_DIRS})
    target_link_libraries(maglev_opensai PRIVATE ${SAI-${component}_LIBRARIES})
endforeach()
target_link_libraries(maglev_opensai PRIVATE Eigen3::Eigen ${CHAI3D_LIBRARIES})
target_compile_definitions(maglev_opensai PRIVATE
    DEMO_CONFIG_FOLDER="${CMAKE_CURRENT_SOURCE_DIR}/xml_config_files"
    DEMO_ROBOT_FOLDER="${DEMO_ROBOT_FOLDER}")
